# Python development on BastiOS, for the compatibility suite (test/compat):
# the root of the official python:alpine image (CPython and pip in
# /usr/local), entered with chroot, with gcc installed by apk from the local
# repository over HTTP. The Python disk (pythondisk.py) holds the root as
# /disk/root.tar and Alpine's BusyBox as /disk/busybox; the web server
# serves the packages of pythondeps.txt at HOSTURL/wheels/ (pip finds them
# through /etc/pip.conf) and the scripts of pyprogs/ at HOSTURL/pyprogs/.
# The cases build on each other and share one guest. The format is
# applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The expected output is what Linux prints for the same commands in a chroot
# of the same root.

== python: unpack the root, chroot into it, run python3
$ bb mkdir -p /disk/a && cd /disk/a && bb tar -xf /disk/root.tar && for d in null:1:3 zero:1:5 random:1:8 urandom:1:9 tty:5:0 console:5:1 ptmx:5:2; do set -- $(echo $d | bb tr : ' '); bb mknod dev/$1 c $2 $3; done; bb mkdir -p dev/pts dev/shm proc tmp root; : > etc/apk/repositories; echo 127.0.0.1 localhost > etc/hosts; bb printf '[global]\nno-index = true\nfind-links = HOSTURL/wheels/\ntrusted-host = 10.0.2.2\n' > etc/pip.conf; cd /; bb mount -t proc proc /disk/a/proc; bb mount -t devpts devpts /disk/a/dev/pts; bb chroot /disk/a /usr/local/bin/python3 -V
Python 3.14.8

== apk: install gcc, musl-dev, binutils and make over HTTP
$ bb chroot /disk/a /sbin/apk add --allow-untrusted -q --no-progress -X ALPINEURL gcc musl-dev binutils make 2>&1 | bb tail -n 3; bb chroot /disk/a /usr/bin/gcc --version | bb head -n 1
gcc (Alpine 15.2.0) 15.2.0

== gcc: build a C program and run it
$ bb printf '#include <stdio.h>\nint main(void) { puts("c 42"); return 0; }\n' > /disk/a/root/h.c && bb chroot /disk/a /bin/sh -c 'cd /root && gcc -O2 -o h h.c && ./h'
c 42

== python: venv and pip
$ bb chroot /disk/a /bin/sh -c 'python3 -m venv /root/v && /root/v/bin/pip --version | cut -d" " -f1,2'
pip 26.2.1

== pip: install pytest and its dependencies from the host's web server
$ bb chroot /disk/a /bin/sh -c '. /root/v/bin/activate && pip install -q pytest 2>&1 | tail -n 3; pytest --version'
pytest 9.1.1

== pip: build a C extension from source (setuptools, gcc) and import it
$ bb chroot /disk/a /bin/sh -c '. /root/v/bin/activate && pip install -q markupsafe 2>&1 | tail -n 3; wget -q -O /root/ext.py HOSTURL/pyprogs/ext.py && python /root/ext.py'
markupsafe._speedups &lt;a&gt; True

== pytest: the whole test suite of a real project (Jinja2's)
$ bb chroot /disk/a /bin/sh -c '. /root/v/bin/activate && pip install -q jinja2 pytest-xdist 2>&1 | tail -n 3; cd /root && wget -q -O - HOSTURL/wheels/jinja2-3.1.6.tar.gz | tar xz && cd jinja2-3.1.6 && python -m pytest -q -n 2 -p no:cacheprovider --ignore=tests/test_async.py --ignore=tests/test_async_filters.py 2>&1 | tail -n 1 | sed "s/ in .*//"'
784 passed

== python: the standard library on top of the system calls (pyprogs/stdlib.py)
$ bb chroot /disk/a /bin/sh -c 'cd /root && wget -q -O stdlib.py HOSTURL/pyprogs/stdlib.py && wget -q -O cert.pem HOSTURL/cert.pem && python3 stdlib.py HOSTSURL /root/cert.pem 2>&1 | grep -v "^ok "'
failed: 0
