# Alpine Linux on BastiOS, for the compatibility suite (test/compat): its
# root file system unpacked on the disk, entered with chroot, and apk
# installing ordinary, dynamically linked Alpine packages from a local
# repository, which then run. The Alpine disk (alpinedisk.py) holds
# Alpine's minirootfs as /disk/root.tar, the repository under
# /disk/a/repo and Alpine's BusyBox as /disk/busybox; the suite boots it
# separately. The format is applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The cases run in order and build on each other: the first unpacks the
# root, the next ones install into it.

== alpine: unpack the root, chroot into it
$ cd /disk/a && bb tar -xf /disk/root.tar && for d in null:1:3 zero:1:5 random:1:8 urandom:1:9 tty:5:0 console:5:1; do set -- $(echo $d | bb tr : ' '); bb mknod dev/$1 c $2 $3; done; echo /repo > etc/apk/repositories; cd /; bb chroot /disk/a /bin/sh -c 'cat /etc/alpine-release; ls /dev | cat'
3.24.2
console
null
random
tty
urandom
zero

== apk: the packages the root came with
$ bb chroot /disk/a /sbin/apk info 2>/dev/null | bb wc -l
16

== apk: install git, curl and less, with what they depend on
$ bb chroot /disk/a /sbin/apk add --allow-untrusted --no-progress git curl less openssh-keygen 2>&1 | bb grep -v -e '^(' -e '^WARNING' -e '^OK:'; echo status $?
Executing busybox-1.37.0-r31.trigger
status 0

== apk: they are in its database
$ bb chroot /disk/a /sbin/apk info -e git curl less openssh-keygen 2>/dev/null
git
curl
less
openssh-keygen

== git: commit, log and diff
$ bb chroot /disk/a /bin/sh -lc 'cd /root && git init -q --template= r && cd r && git config user.name a && git config user.email a@b && printf "a\nb\n" > f && git add f && git commit -qm one && printf "a\nc\n" > f && git commit -qam two && git --no-pager log --format=%s && git diff --stat HEAD~1 | cat'
two
one
 f | 2 +-
 1 file changed, 1 insertion(+), 1 deletion(-)

== git: clone, gc and fsck
$ bb chroot /disk/a /bin/sh -lc 'cd /root && git clone -q --template= r r2 && cd r2 && git --no-pager log --oneline | wc -l && git gc -q && git fsck --no-progress 2>&1 && ls .git/objects/pack | wc -l; cd /root && rm -rf r r2'
2
3

== curl: a file from the host
$ bb chroot /disk/a /usr/bin/curl -s HOSTURL/hello.txt
hello from the host

== curl: 1 MiB from the host
$ bb chroot /disk/a /bin/sh -lc 'curl -s -o /root/m.bin HOSTURL/mib.bin && md5sum /root/m.bin; rm /root/m.bin'
85bfb9943e6f0cb09f0f52679e9eac2b  /root/m.bin

== ssh-keygen: an ed25519 key
$ bb chroot /disk/a /bin/sh -lc 'ssh-keygen -q -t ed25519 -N "" -C k -f /root/k && ssh-keygen -l -f /root/k | cut -d" " -f1,4; head -c 11 /root/k.pub; echo; rm /root/k /root/k.pub'
256 (ED25519)
ssh-ed25519

== less: copies a pipe through
$ printf 'one\ntwo\n' | bb chroot /disk/a /usr/bin/less -FX > /disk/a/root/o; bb cat /disk/a/root/o; rm /disk/a/root/o
one
two

== apk: remove a package and its files
$ bb chroot /disk/a /sbin/apk del --no-progress -q less 2>/dev/null; echo status $?; bb chroot /disk/a /sbin/apk info -e less 2>/dev/null; test -e /disk/a/usr/bin/less || echo gone
status 0
gone
