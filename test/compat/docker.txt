# Docker on BastiOS, for the compatibility suite (test/compat): the
# docker:dind image's dockerd (with its built-in containerd) and runc
# entered with chroot, as root. The Docker disk (dockerdisk.py) holds the
# image's root as a tar file (/disk/root.tar) and a BusyBox at
# /disk/busybox. The storage driver is vfs (no kernel overlay file system
# for it) and there is no iptables: the cases that name no network run on
# the default bridge (docker0; the kernel masquerades for it, and -p goes
# through docker-proxy), the rest with --network=none. Docker's embedded DNS
# server starts only if its iptables commands succeed, so iptables is a script
# that does nothing: the kernel does the redirect (docs/linux.md).
# The format is applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The expected output is what Linux prints for the same commands in a chroot
# of the same root.

== docker: unpack the image root, start dockerd and wait for it
$ bb() { /disk/busybox "$@"; }; bb mkdir -p /disk/p && cd /disk/p && bb tar -xf /disk/root.tar && for t in iptables ip6tables; do bb rm -f usr/sbin/$t; echo '#!/bin/sh' > usr/sbin/$t; echo 'exit 0' >> usr/sbin/$t; bb chmod +x usr/sbin/$t; done && bb mkdir -p dev proc sys tmp var/tmp run root dev/shm dev/pts var/lib/docker && for d in null:1:3 zero:1:5 full:1:7 random:1:8 urandom:1:9 tty:5:0 console:5:1; do set -- $(echo $d | bb tr : ' '); bb mknod -m 666 dev/$1 c $2 $3; done; cd /; bb mount -t proc proc /disk/p/proc; bb mount -t sysfs sysfs /disk/p/sys; bb mkdir -p /disk/p/sys/fs/cgroup; bb mount -t cgroup2 cgroup2 /disk/p/sys/fs/cgroup; bb mount -t tmpfs tmpfs /disk/p/dev/shm; bb mount -t devpts devpts /disk/p/dev/pts; bb chmod 1777 /disk/p/tmp; bb mount --make-rshared /; bb chroot /disk/p /bin/sh -c 'dockerd --iptables=false --ip6tables=false --storage-driver=vfs --feature containerd-snapshotter=false > /tmp/dockerd.log 2>&1 &'; n=0; until bb chroot /disk/p /usr/local/bin/docker version >/dev/null 2>&1 || [ $n -ge 90 ]; do bb sleep 2; n=$((n+1)); done; dc() { bb chroot /disk/p /usr/local/bin/docker "$@"; }; dc version --format '{{.Server.Os}}'
linux

== docker: import a root and run a container
$ bb tar -cf /disk/p/tmp/ctr.tar -C /disk/p bin lib && dc import /tmp/ctr.tar ctr >/dev/null 2>&1; dc run --rm --network=none ctr echo hello 2>&1
hello

== docker: the exit status of a container comes back
$ dc run --rm --network=none ctr sh -c 'echo out; exit 3' 2>/dev/null; echo rc=$?
out
rc=3

== docker: a container has its own hostname
$ dc run --rm --network=none --hostname boxed ctr hostname 2>&1
boxed

== docker: a detached container's output is in its log
$ dc run -d --network=none --name logger ctr sh -c 'echo logged' >/dev/null 2>&1; dc wait logger >/dev/null 2>&1; dc logs logger 2>&1
logged

== docker: a memory limit reaches the container's cgroup
$ dc run --rm --network=none --memory 64m ctr cat /sys/fs/cgroup/memory.max 2>&1
67108864

== docker: a container on the default bridge has an address and a default route
$ dc run --rm ctr /bin/busybox sh -c '/bin/busybox ip route | /bin/busybox sed "s/ *src.*//; s/ *$//"' 2>&1
default via 172.17.0.1 dev eth0
172.17.0.0/16 dev eth0 scope link

== docker: a container reaches the bridge address
$ dc run --rm ctr /bin/busybox ping -c 1 -W 5 172.17.0.1 2>&1 | bb grep -c '1 packets received'
1

== docker: two containers talk over the bridge, and the host reaches them
$ dc run -d --name srv ctr /bin/busybox sh -c 'echo from-srv | /bin/busybox nc -l -p 8080' >/dev/null 2>&1; ip=$(dc inspect -f '{{.NetworkSettings.Networks.bridge.IPAddress}}' srv); dc run --rm ctr /bin/busybox nc -w 10 $ip 8080 2>&1; bb ping -c 1 -W 5 $ip 2>&1 | bb grep -c '1 packets received'; dc rm -f srv >/dev/null 2>&1
from-srv
1

== docker: a container on the bridge reaches the host's network
$ dc run --rm ctr /bin/busybox wget -q -O - -T 10 HOSTURL/hello.txt 2>&1
hello from the host

== docker: a published port reaches the container
$ dc run -d --name pub -p 8082:8080 ctr /bin/busybox sh -c 'echo from-pub | /bin/busybox nc -l -p 8080' >/dev/null 2>&1; bb sleep 2; bb nc -w 5 127.0.0.1 8082 </dev/null | bb head -1; dc rm -f pub >/dev/null 2>&1
from-pub

== docker: two containers on a user-defined network reach each other by address
$ dc network create nn >/dev/null 2>&1; dc run -d --name nsrv --network nn ctr /bin/busybox sleep 120 >/dev/null 2>&1; ip=$(dc inspect -f '{{.NetworkSettings.Networks.nn.IPAddress}}' nsrv); dc run --rm --network nn ctr /bin/busybox ping -c 1 -W 5 $ip 2>&1 | bb grep -c '1 packets received'; dc rm -f nsrv >/dev/null 2>&1; dc network rm nn >/dev/null 2>&1
1

== docker: containers on a user-defined network resolve each other's names
$ dc network create nn2 >/dev/null 2>&1; dc run -d --name dsrv --network nn2 ctr /bin/busybox sleep 120 >/dev/null 2>&1; dc run --rm --network nn2 ctr /bin/busybox ping -c 1 -W 5 dsrv 2>&1 | bb grep -c '1 packets received'; dc rm -f dsrv >/dev/null 2>&1; dc network rm nn2 >/dev/null 2>&1
1
