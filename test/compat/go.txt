# The Go toolchain on BastiOS, for the compatibility suite (test/compat): the
# compiler, linker and standard library of the official golang:alpine image,
# entered with chroot. Go programs are static, so the root holds only
# /usr/local/go. The Go disk (godisk.py) holds it as /disk/g and a BusyBox as
# /disk/busybox. The format is applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The expected output is what Linux prints for the same commands in a chroot
# of the same root.

$ cd /disk/g && bb mkdir -p dev proc tmp etc work; for d in null:1:3 zero:1:5 random:1:8 urandom:1:9 tty:5:0 console:5:1; do set -- $(echo $d | bb tr : ' '); bb mknod dev/$1 c $2 $3; done; cd /; bb ls /disk/g/dev | bb cat; bb mount -t proc proc /disk/g/proc

== go: goroutines, channels and the collector
$ bb printf 'package main\nimport ("fmt"; "runtime"; "sync")\nfunc main() {\n var wg sync.WaitGroup\n ch := make(chan int, 100)\n for i := 0; i < 100; i++ { wg.Add(1); go func(n int) { defer wg.Done(); b := make([]byte, 1<<16); b[0] = byte(n); ch <- int(b[0]) }(i) }\n wg.Wait(); close(ch)\n sum := 0\n for v := range ch { sum += v }\n runtime.GC()\n var m runtime.MemStats\n runtime.ReadMemStats(&m)\n fmt.Println("sum", sum, "gc", m.NumGC > 0)\n}\n' > /disk/g/work/g.go; export GOCACHE=/tmp/gc HOME=/tmp CGO_ENABLED=0 GOTOOLCHAIN=local GO111MODULE=off GOFLAGS=-p=1; bb chroot /disk/g /usr/local/go/bin/go run /work/g.go
sum 4950 gc true

== go: files, directories and the clock

== go: gopls reports a type error
$ bb mkdir -p /disk/g/work/ls; bb printf 'module ls\ngo 1.24\n' > /disk/g/work/ls/go.mod; bb printf 'package main\n\nfunc add(a, b int) int { return a + b }\n\nfunc main() {\n\tvar s string = add(1, 2)\n\t_ = s\n}\n' > /disk/g/work/ls/main.go; export GOCACHE=/tmp/gc HOME=/tmp CGO_ENABLED=0 GOTOOLCHAIN=local GOFLAGS=-p=1 GOPATH=/tmp/gp PATH=/usr/local/go/bin:/usr/local/bin; bb chroot /disk/g /usr/local/bin/gopls check /work/ls/main.go 2>&1 | bb sed 's/^[^:]*://'
6:17-26: cannot use add(1, 2) (value of type int) as string value in variable declaration

== go: gopls finds the definition of a function
