# The second X11 disk (x11disk.py --part2), for the apps added after the first
# one's roots used up the 262144 entries a BastiOS disk holds: the server and
# client roots again, and one root per app. The format is x11.txt's.

== x11b: unpack the server and the client roots
$ bb mkdir -p /disk/x /disk/t /tmp && bb tar -xf /disk/server.tar -C /disk/x && bb tar -xf /disk/client.tar -C /disk/t && for r in x t; do bb mkdir -p /disk/$r/dev /disk/$r/proc /disk/$r/tmp/.X11-unix && bb mount --bind /dev /disk/$r/dev && bb mount -t proc proc /disk/$r/proc; done; bb mkdir -p /disk/x/tmp/fb; echo ok
ok
