# The Rust toolchain on BastiOS, for the compatibility suite (test/compat):
# rustc and cargo of the official rust:alpine image, with Alpine's gcc and
# musl, which rustc links with, entered with chroot. The Rust disk
# (rustdisk.py) holds the image's root as a tar file (/disk/root.tar) and a
# BusyBox at /disk/busybox. The format is applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The expected output is what Linux prints for the same commands in a chroot
# of the same root.

== rust: collections, strings, closures and traits
$ bb printf 'use std::collections::HashMap;\ntrait Area { fn area(&self) -> f64; }\nstruct Sq(f64);\nstruct Ci(f64);\nimpl Area for Sq { fn area(&self) -> f64 { self.0 * self.0 } }\nimpl Area for Ci { fn area(&self) -> f64 { 3.0 * self.0 * self.0 } }\nfn main() {\n  let mut m: HashMap<String, usize> = HashMap::new();\n  for w in "the cat and the dog and the bird".split_whitespace() { *m.entry(w.to_string()).or_insert(0) += 1; }\n  let mut v: Vec<_> = m.into_iter().collect();\n  v.sort_by(|a, b| b.1.cmp(&a.1).then(a.0.cmp(&b.0)));\n  println!("{:?}", &v[..3]);\n  let shapes: Vec<Box<dyn Area>> = vec![Box::new(Sq(2.0)), Box::new(Ci(1.0))];\n  println!("{}", shapes.iter().map(|s| s.area()).sum::<f64>());\n  let sq: Vec<u64> = (1..=5).map(|x| x * x).collect();\n  println!("{:?} {}", sq, sq.iter().sum::<u64>());\n}\n' > /disk/r/work/c.rs; rc rustc -O /work/c.rs -o /work/c && rc /work/c
[("the", 3), ("and", 2), ("bird", 1)]
7
[1, 4, 9, 16, 25] 55

== rust: threads, channels and a mutex
$ bb printf 'use std::sync::{mpsc, Arc, Mutex};\nuse std::thread;\nfn main() {\n  let n = Arc::new(Mutex::new(0u64));\n  let (tx, rx) = mpsc::channel();\n  let hs: Vec<_> = (0..4).map(|i| { let n = n.clone(); let tx = tx.clone(); thread::spawn(move || { for _ in 0..1000 { *n.lock().unwrap() += 1; } tx.send(i).unwrap(); }) }).collect();\n  drop(tx);\n  for h in hs { h.join().unwrap(); }\n  let mut got: Vec<i32> = rx.iter().collect();\n  got.sort();\n  println!("{} {:?}", *n.lock().unwrap(), got);\n}\n' > /disk/r/work/t.rs; rc rustc -O /work/t.rs -o /work/t && rc /work/t
4000 [0, 1, 2, 3]
