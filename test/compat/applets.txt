# BusyBox applet smoke tests for the compatibility suite (test/compat).
#
#   == NAME
#   $ COMMAND        (one line, run by ash; `bb` runs the suite's BusyBox)
#   expected output, line by line, compared exactly
#
# Commands run in /, which is BastiOS's RAM file system. Keep files they
# make under names starting with "ca_" and remove them in the same command.
# Do not use /dev/null or /tmp here: whether those exist is measured by the
# C programs, and a case should measure its applet.

== echo
$ bb echo hello world
hello world

== printf
$ bb printf '%s-%03d\n' a 7
a-007

== true/false
$ bb true && bb false || echo ok
ok

== test
$ bb test 3 -gt 2 && echo yes
yes

== seq
$ bb seq 3
1
2
3

== expr
$ bb expr 6 \* 7
42

== basename/dirname
$ bb basename /a/b/c.txt .txt; bb dirname /a/b/c.txt
c
/a/b

== cat a file
$ bb echo cat-me > ca_1; bb cat ca_1; bb rm ca_1
cat-me

== wc
$ bb printf 'a b\nc\n' | bb wc -l
2

== head/tail
$ bb seq 10 | bb head -n 3 | bb tail -n 1
3

== sort -n
$ bb printf '10\n9\n100\n' | bb sort -n
9
10
100

== uniq -c
$ bb printf 'a\na\nb\n' | bb uniq -c | bb sed 's/^ *//'
2 a
1 b

== cut
$ echo a:b:c | bb cut -d: -f2
b

== tr
$ echo hello | bb tr a-z A-Z
HELLO

== grep
$ bb printf 'one\ntwo\nthree\n' | bb grep -n t
2:two
3:three

== sed
$ echo hello | bb sed 's/l/L/g'
heLLo

== awk
$ bb printf '1 2\n3 4\n' | bb awk '{s += $1 * $2} END {print s}'
14

== od
$ bb printf 'AB' | bb od -An -tx1
 41 42

== hexdump
$ bb printf 'AB' | bb hexdump -C | bb head -n 1
00000000  41 42                                             |AB|

== md5sum
$ bb printf abc | bb md5sum
900150983cd24fb0d6963f7d28e17f72  -

== sha256sum
$ bb printf abc | bb sha256sum
ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad  -

== base64
$ bb printf hello | bb base64
aGVsbG8=

== yes | head
$ bb yes x | bb head -n 2
x
x

== tee
$ echo tee | bb tee ca_t > ca_u; bb cat ca_t; bb rm ca_t ca_u
tee

== xargs
$ bb printf 'a\nb\n' | bb xargs bb echo
a b

== env
$ X=1 bb env | bb grep '^X='
X=1

== lang
$ bb env | bb grep '^LANG='
LANG=C.UTF-8

== printenv
$ Y=2 bb printenv Y
2

== uname
$ bb uname -s
Linux

== id -u
$ bb id -u
0

== whoami
$ bb whoami
root

== hostname
$ bb hostname | bb grep -c .
1

== nproc
$ bb nproc | bb grep -c '^[0-9]'
1

== date
$ bb date +%Y | bb grep -c '^20'
1

== date -d
$ bb date -u -d @0 +%Y-%m-%d
1970-01-01

== sleep
$ bb sleep 0.2 && echo slept
slept

== usleep
$ bb usleep 1000 && echo ok
ok

== touch/ls
$ bb touch ca_x && bb ls ca_x && bb rm ca_x
ca_x

== ls -l
$ bb echo 12345 > ca_l; bb ls -l ca_l | bb awk '{print $5, $9}'; bb rm ca_l
6 ca_l

== cp
$ echo cp > ca_a; bb cp ca_a ca_b; bb cat ca_b; bb rm ca_a ca_b
cp

== mv
$ echo mv > ca_a; bb mv ca_a ca_b; bb cat ca_b; bb test -e ca_a && echo left; bb rm ca_b
mv

== ln -s
$ echo ln > ca_a; bb ln -s ca_a ca_b && bb cat ca_b; bb rm -f ca_a ca_b
ln

== readlink/realpath
$ bb realpath /
/

== stat
$ echo 1234 > ca_s; bb stat -c %s ca_s; bb rm ca_s
5

== chmod
$ echo x > ca_c; bb chmod 700 ca_c && bb stat -c %a ca_c; bb rm ca_c
700

== mkdir/rmdir
$ bb mkdir ca_d && bb rmdir ca_d && echo ok
ok

== mkdir -p
$ bb mkdir -p ca_d/e/f && bb rm -r ca_d && echo ok
ok

== find
$ bb mkdir ca_d; bb touch ca_d/x; bb find ca_d -name x; bb rm -r ca_d
ca_d/x

== du
$ bb du /disk/busybox | bb grep -c busybox
1

== df
$ bb df / | bb grep -c /
1

== dd
$ bb dd if=/dev/zero bs=512 count=2 2>ca_e | bb wc -c; bb rm ca_e
1024

== dd large blocks
$ bb dd if=/dev/zero bs=65536 count=16 2>ca_e | bb wc -c; bb head -n 1 ca_e; bb rm ca_e
1048576
16+0 records in

== dd one large read of a file
$ bb dd if=/dev/zero of=ca_big bs=65536 count=40 2>/dev/null; bb dd if=ca_big bs=2097152 count=1 2>ca_e | bb wc -c; bb head -n 1 ca_e; bb rm ca_big ca_e
2097152
1+0 records in

== cmp
$ echo a > ca_1; echo a > ca_2; bb cmp ca_1 ca_2 && echo same; bb rm ca_1 ca_2
same

== diff
$ echo a > ca_1; echo b > ca_2; bb diff ca_1 ca_2 | bb tail -n 2; bb rm ca_1 ca_2
-a
+b

== gzip round trip
$ echo squeeze | bb gzip | bb gunzip
squeeze

== tar round trip
$ echo t > ca_1; bb tar cf ca_t ca_1 && bb rm ca_1 && bb tar xf ca_t && bb cat ca_1; bb rm -f ca_1 ca_t
t

== mktemp
$ f=$(bb mktemp) && bb rm "$f" && echo ok
ok

== which
$ PATH=/bin bb which ls
/bin/ls

== timeout
$ bb sh -c 'bb timeout 1 bb sleep 5; echo $?' 2>ca_e; bb rm ca_e
143

== nice
$ bb nice -n 1 bb echo niced
niced

== time
$ bb time bb true 2>&1 | bb grep -c real
1

== sync
$ bb sync && echo ok
ok

== ps
$ bb ps | bb head -n 1 | bb grep -c PID
1

== kill -0
$ bb kill -0 $$ && echo alive
alive

== free
$ bb free | bb grep -c Mem
1

== uptime
$ bb uptime | bb grep -c load
1

== sh -c
$ bb sh -c 'x=4; echo $((x * x))'
16

== sh pipeline and subshell
$ bb sh -c '(echo a; echo b) | bb wc -l'
2
