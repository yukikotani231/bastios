# The JDK on BastiOS, for the compatibility suite (test/compat): java, javac
# and jar of the official eclipse-temurin alpine image, entered with chroot.
# The JDK disk (jdkdisk.py) holds the image's root as a tar file
# (/disk/root.tar) and a BusyBox at /disk/busybox. The format is
# applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The expected output is what Linux prints for the same commands in a chroot
# of the same root. The JDK reads its modules file (lib/modules, 130 MB) with
# pread and a mapping; a read that came back short, or a page that read as
# zeros, shows as a ClassFormatError.

== java: records, streams and collections
$ bb printf 'public class Rec { record P(String n, int a) {} public static void main(String[] x) { java.util.List<P> l = java.util.List.of(new P("bo", 31), new P("al", 25), new P("cy", 40)); System.out.println(l.stream().sorted(java.util.Comparator.comparingInt(P::a)).map(P::n).collect(java.util.stream.Collectors.joining(","))); System.out.println(l.stream().mapToInt(P::a).sum() + " " + l.get(1)); } }\n' > /disk/r/work/Rec.java; jv -d /work /work/Rec.java && jj -cp /work Rec

== java: threads and an executor
$ bb printf 'import java.util.concurrent.*; import java.util.concurrent.atomic.*;\npublic class T { public static void main(String[] x) throws Exception {\n  ExecutorService ex = Executors.newFixedThreadPool(4); AtomicLong n = new AtomicLong(); java.util.List<Future<Integer>> fs = new java.util.ArrayList<>();\n  for (int i = 0; i < 4; i++) { final int k = i; fs.add(ex.submit(() -> { for (int j = 0; j < 1000; j++) n.incrementAndGet(); return k; })); }\n  int s = 0; for (Future<Integer> f : fs) s += f.get(); ex.shutdown();\n  System.out.println(n.get() + " " + s); } }\n' > /disk/r/work/T.java; jv -d /work /work/T.java && jj -cp /work T
4000 6
