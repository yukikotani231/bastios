# Database servers on BastiOS, for the compatibility suite (test/compat): the
# PostgreSQL, Redis and Valkey servers of the official Alpine-based images,
# each unpacked from a tar file on the database disk (dbdisk.py) and entered
# with chroot. The servers are dynamically linked musl programs; PostgreSQL
# runs as its own user (BusyBox su) with several processes, shared memory,
# fsync and a write-ahead log, Redis and Valkey fork to save. The format is
# applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The cases share one shell, so the first defines functions the others use.
# The expected output is what Linux prints for the same commands in a chroot
# of the same root.

== db: unpack the roots and make the device nodes
$ for n in p:postgres r:redis v:valkey; do d=${n%%:*}; bb mkdir -p /disk/$d && cd /disk/$d && bb tar -xf /disk/${n#*:}.tar && for x in null:1:3 zero:1:5 random:1:8 urandom:1:9 tty:5:0; do set -- $(echo $x | bb tr : ' '); bb mknod dev/$1 c $2 $3; done; done; cd /; bb mount -t proc proc /disk/p/proc; pgsu() { bb chroot /disk/p /bin/su postgres -s /bin/sh -c "$1"; }; rcli() { bb chroot /disk/r /usr/local/bin/redis-cli "$@"; }; vcli() { bb chroot /disk/v /usr/local/bin/valkey-cli "$@"; }; bb ls /disk/p/dev | bb cat
null
random
tty
urandom
zero

== db: postgres recovers a committed row after the server is killed
