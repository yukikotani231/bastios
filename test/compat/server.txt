# Servers on BastiOS, for the compatibility suite (test/compat): Ubuntu's
# Redis, memcached and nginx, linked against glibc, entered with chroot and
# talking to their own clients over the loopback interface. The server disk
# (serverdisk.py) holds the root as /disk/root.tar and a BusyBox as
# /disk/busybox. The format is applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The cases run in order: the first unpacks the root.

== server: unpack the root, make the device nodes
$ mkdir -p /disk/s && cd /disk/s && bb tar -xf /disk/root.tar && for d in null:1:3 zero:1:5 random:1:8 urandom:1:9 tty:5:0; do set -- $(echo $d | bb tr : ' '); bb mknod dev/$1 c $2 $3; done; bb chmod 666 dev/*; printf 'root:x:0:0::/:/bin/sh\nnobody:x:65534:65534::/:/bin/false\n' > etc/passwd; printf 'root:x:0:\nnobody:x:65534:\nnogroup:x:65534:\n' > etc/group; bb mkdir -p proc; bb mount -t proc proc /disk/s/proc; cd /; bb chroot /disk/s /usr/bin/redis-server --version | bb cut -c1-30; bb chroot /disk/s /usr/bin/memcached --version; bb chroot /disk/s /usr/sbin/nginx -v 2>&1
Redis server v=7.0.15 sha=0000
memcached 1.6.24
nginx version: nginx/1.24.0 (Ubuntu)

== server: redis restarts and loads the snapshot
$ R="bb chroot /disk/s /usr/bin/redis-cli"; $R shutdown nosave; bb sleep 1; bb chroot /disk/s /usr/bin/redis-server --port 6379 --bind 127.0.0.1 --daemonize yes --dir /tmp --logfile /tmp/redis.log --save '' >/dev/null 2>&1; bb sleep 2; $R get persist; $R shutdown nosave
"yes"

== server: nginx serves a page

== server: redis replicates to a second server
$ R="bb chroot /disk/s /usr/bin/redis-cli"; bb mkdir -p /disk/s/tmp/r1 /disk/s/tmp/r2; bb chroot /disk/s /usr/bin/redis-server --port 6379 --bind 127.0.0.1 --daemonize yes --dir /tmp/r1 --logfile /tmp/redis.log --save '' --repl-diskless-sync-delay 0 >/dev/null 2>&1; bb sleep 2; $R set shared one >/dev/null; bb chroot /disk/s /usr/bin/redis-server --port 6380 --bind 127.0.0.1 --daemonize yes --dir /tmp/r2 --logfile /tmp/redis2.log --save '' --replicaof 127.0.0.1 6379 >/dev/null 2>&1; bb sleep 4; $R -p 6380 get shared; $R -p 6380 info replication | bb grep -e '^role' -e master_link_status | bb tr -d '\r'; $R set later two >/dev/null; bb sleep 1; $R -p 6380 get later; $R -p 6380 shutdown nosave; $R shutdown nosave

role:slave
master_link_status:up
"two"

== server: nginx runs workers under a master

hello from nginx
2
status 1
