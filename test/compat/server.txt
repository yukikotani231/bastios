# Server programs on BastiOS, for the compatibility suite (test/compat): the
# nginx, redis and PostgreSQL of the official Docker images (their Alpine
# variants), each entered with chroot and run the way it is run in production:
# a web server serving files and acting as a reverse proxy, a key-value store
# with persistence and a replica, a SQL database that is created, loaded,
# dumped, restarted and recovered after a crash. The server disk
# (serverdisk.py) holds each image's root as a tar file (/disk/nginx.tar,
# redis.tar, postgres.tar) and a BusyBox at /disk/busybox; the suite boots
# it separately. The format is applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The cases run in order and build on each other. The clients (wget, redis-cli,
# psql, pgbench) run in the guest too, over the loopback.

== nginx: unpack the root, make the device nodes
$ bb mkdir -p /disk/ng && cd /disk/ng && bb tar -xf /disk/nginx.tar && bb mkdir -p dev proc tmp www bk && for d in null:1:3 zero:1:5 random:1:8 urandom:1:9 tty:5:0 console:5:1; do set -- $(echo $d | bb tr : ' '); bb mknod dev/$1 c $2 $3; done; cd /; bb chroot /disk/ng /usr/sbin/nginx -v 2>&1

== nginx: a site, a proxy and a backend, both configurations valid
$ bb seq 1 300000 > /disk/ng/www/big.txt; cp /disk/ng/www/big.txt /disk/ng/bk/big.txt; echo hello static > /disk/ng/www/index.html; echo hello backend > /disk/ng/bk/index.html; printf 'worker_processes 2;\ndaemon on;\npid /run/nginx.pid;\nerror_log /tmp/err.log;\nevents { worker_connections 256; }\nhttp {\n include /etc/nginx/mime.types;\n access_log /tmp/acc.log;\n sendfile on;\n server {\n  listen 8080;\n  listen [::]:8080 ipv6only=on;\n  root /www;\n  location /api/ { proxy_pass http://127.0.0.1:9000/; }\n }\n}\n' > /disk/ng/etc/nginx/nginx.conf; printf 'worker_processes 1;\ndaemon on;\npid /run/be.pid;\nerror_log /tmp/be_err.log;\nevents { worker_connections 64; }\nhttp {\n access_log off;\n server { listen 9000; root /bk; }\n}\n' > /disk/ng/etc/nginx/be.conf; bb chroot /disk/ng /usr/sbin/nginx -t 2>&1 | bb tail -n 1; bb chroot /disk/ng /usr/sbin/nginx -t -c /etc/nginx/be.conf 2>&1 | bb tail -n 1
