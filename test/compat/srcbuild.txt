# Real projects built from their source tarballs inside Alpine (BastiOS's own
# gcc, make, ld and the shell, running a configure script and hundreds of
# compiler runs), then run. The tarballs are the upstream releases, pinned by
# SHA-256 (srcpkg.sh) and served at HOSTURL/src. Own lane: a copy of the
# Alpine disk, where the cases run in order and build on each other. The
# format is applets.txt's:
#
#   == NAME
#   $ COMMAND        (one line, run by ash in /)
#   expected output, line by line, compared exactly
#
# The Alpine repository has no -dev packages, so zlib is built first and git
# and curl link against it.

== srcbuild: unpack the Alpine root, install gcc, make and the kernel headers
$ cd /disk/a && bb tar -xf /disk/root.tar && for d in null:1:3 zero:1:5 random:1:8 urandom:1:9 tty:5:0 console:5:1; do set -- $(echo $d | bb tr : ' '); bb mknod dev/$1 c $2 $3; done; echo /repo > etc/apk/repositories; cd /; bb chroot /disk/a /sbin/apk add --allow-untrusted -q --no-progress -X ALPINEURL gcc musl-dev make linux-headers 2>/dev/null; bb chroot /disk/a /sbin/apk info -e gcc musl-dev make linux-headers 2>/dev/null

== git 2.53: make install, then commit, clone, gc and fsck
$ bb chroot /disk/a /bin/sh -c 'cd /root && tar xJf git-2.53.0.tar.xz && cd git-2.53.0 && make -j2 prefix=/root/inst NO_REGEX=NeedsStartEnd NO_GETTEXT=1 NO_EXPAT=1 NO_TCLTK=1 NO_PERL=1 NO_PYTHON=1 NO_CURL=1 NO_OPENSSL=1 CFLAGS=-O0 install >/root/git.log 2>&1 && export PATH=/root/inst/bin:$PATH HOME=/root && cd /root && git --version && git init -q --template= r && cd r && git config user.name a && git config user.email a@b && printf "a\nb\n" > f && git add f && git commit -qm one && printf "a\nc\n" > f && git commit -qam two && git --no-pager log --format=%s && cd .. && git clone -q --template= r r2 && cd r2 && git gc -q && git fsck --no-progress 2>&1; git --no-pager log --oneline | wc -l; cd /root && rm -rf git-2.53.0 inst r r2'

== busybox 1.37 (the compat suite's applet set): allnoconfig, make, and BusyBox's own testsuite for sed, grep, sort, cut and tr
$ bb chroot /disk/a /bin/sh -c 'cd /root && mkdir bbr && tar xjf busybox-1.37.0.tar.bz2 -C bbr --strip-components=1 && cd bbr && make allnoconfig >/dev/null 2>&1; for o in $(grep -v "^#" /root/busybox.config); do sed -i "s/^# $o is not set/$o=y/" .config; done; yes "" | make oldconfig >/dev/null 2>&1; make -j2 >/root/bbr.log 2>&1 && ./busybox sh -c "echo built \$((6*7))" && cd testsuite && for a in sed grep sort cut tr; do SKIP_KNOWN_BUGS=1 bindir=/root/bbr ./runtest $a; done > /root/bbt.log 2>&1; echo "failed: $(grep -c "^FAIL" /root/bbt.log)"; test $(grep -c "^PASS" /root/bbt.log) -ge 150 && echo "passed: 150 or more"; cd /root && rm -rf bbr'
